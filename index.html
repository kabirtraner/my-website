ID: SC-042
Scenario Challenge: Optimize structural image build steps to fire concurrently across multiple Node target microservices running variant semantic versions while isolating production artifacts using custom staging runners.
deploy-pipeline.yaml 
name: Multi-Environment Microservice Engine
on:
  push:
    branches: [ main, release/* ]

jobs:
  orchestrate-matrix:
    runs-on: ubuntu-latest
    strategy:
      matrix:
        service: [auth-api, payment-gateway, inventory-worker]
        node-version: [18.x, 20.x]
    
    steps:
      - name: Source Check-out
        uses: actions/checkout@v4

      - name: Initialize Runtime Environment v${{ matrix.node-version }}
        uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node-version }}
          cache: 'npm'

      - name: Parallel Image Composition & Compilation
        run: |
          cd services/${{ matrix.service }}
          npm ci
          npm run build --if-present
